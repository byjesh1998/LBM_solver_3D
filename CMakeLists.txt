cmake_minimum_required(VERSION 3.14)
project(lbm_squirmer CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
file(GLOB UTILS_SRC ${CMAKE_SOURCE_DIR}/src/utils/*.cpp)
add_executable(lbm_squirmer src/main.cpp ${UTILS_SRC})
target_compile_options(lbm_squirmer PRIVATE -Wall -Wextra $<$<CONFIG:Release>:-O3 -march=native>)
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(lbm_squirmer PRIVATE OpenMP::OpenMP_CXX)
endif()
set_target_properties(lbm_squirmer PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/bin)
